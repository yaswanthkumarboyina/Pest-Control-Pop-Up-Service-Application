const propertyRadios = document.querySelectorAll('input[name="property_type"]');
const serviceSelect = document.getElementById("serviceSelect");
const quantityField = document.getElementById("quantityField");
const quantityInput = document.getElementById("quantityInput");
const totalPriceEl = document.getElementById("totalPrice");
const responseMessage = document.getElementById("responseMessage");

let currentServiceList = {};
let currentProductMap = {};
let selectedVariantId = null;

// Update service dropdown
propertyRadios.forEach(radio => {
  radio.addEventListener("change", () => {
    const selected = radio.value;
    currentServiceList = selected === "Residential" ? residentialServices : commercialServices;
    currentProductMap = selected === "Residential" ? residentialProductMap : commercialProductMap;

    serviceSelect.innerHTML = `<option value="">-- Select Service --</option>`;
    for (const [name] of Object.entries(currentServiceList)) {
      const option = document.createElement("option");
      option.value = name;
      option.textContent = name;
      serviceSelect.appendChild(option);
    }

    totalPriceEl.textContent = "₹0";
    quantityField.style.display = "none";
    quantityInput.value = 1;
    selectedVariantId = null;
  });
});

// Update total price
function updatePrice() {
  const serviceName = serviceSelect.value;
  const service = currentServiceList[serviceName];
  if (!service) {
    totalPriceEl.textContent = "₹0";
    quantityField.style.display = "none";
    selectedVariantId = null;
    return;
  }

  selectedVariantId = currentProductMap[serviceName];

  if (service.pricePerSqft) {
    quantityField.style.display = "block";
    const qty = parseInt(quantityInput.value) || 1;
    totalPriceEl.textContent = "₹" + (qty * service.pricePerSqft).toLocaleString();
  } else {
    quantityField.style.display = "none";
    totalPriceEl.textContent = service.price ? "₹" + service.price.toLocaleString() : "₹0";
  }
}

serviceSelect.addEventListener("change", updatePrice);
quantityInput.addEventListener("input", updatePrice);

    pinCodeError.textContent = "⚠️ Sorry, this service is not available in your area.";
    pinCodeInput.style.borderColor = "red";
    agriPopupForm.style.display = "none";
  } else {

    agriPopupForm.style.display = "block";
  }

// Form submission with Shopify AJAX
document.getElementById("pestControlForm").addEventListener("submit", async function(e) {
  e.preventDefault();
  responseMessage.textContent = "";

  const formData = new FormData(e.target);
  const propertyType = formData.get("property_type");
  const serviceName = formData.get("service");

  if (!propertyType || !serviceName) {
    alert("Please select both Property Type and Service.");
    return;
  }

  const variantId = selectedVariantId;
  if (!variantId) {
    alert("This service is not available for cart checkout.");
    return;
  }

  // Validate minimum 500 sqft for specific products
  const qty = parseInt(quantityInput.value) || 1;
  if (min500Products.includes(variantId) && qty < 500) {
    const areaError = document.getElementById("areaError");
    areaError.textContent = "Please enter a minimum of 500 sqft";
    quantityInput.style.borderColor = "red";
    return;
  } else {
    document.getElementById("areaError").textContent = "";
    quantityInput.style.borderColor = "";
  }

  let cartQuantity = 1;
  const service = currentServiceList[serviceName];
  if (service.pricePerSqft) {
    cartQuantity = qty;
  }

  const attributes = {};
  formData.forEach((value, key) => attributes[key] = value);
  attributes["Total Price"] = totalPriceEl.textContent;

  const payload = { id: variantId, quantity: cartQuantity, properties: attributes };

  try {
    const response = await fetch("/cart/add.js", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload)
    });

    if (response.ok) {
      responseMessage.style.color = "#328130";
      responseMessage.textContent = `✅ ${cartQuantity} item(s) added to your cart successfully!`;
      setTimeout(() => (window.location.href = "/cart"), 1200);
    } else {
      const error = await response.json();
      responseMessage.style.color = "red";
      responseMessage.textContent = "⚠️ Error: " + (error.description || "Unable to add service.");
    }
  } catch (err) {
    console.error(err);
    responseMessage.style.color = "red";
    responseMessage.textContent = "⚠️ Something went wrong. Please try again.";
  }
});
